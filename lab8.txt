CREATE OR REPLACE FUNCTION get_job_title (
    p_job_id IN jobs.job_id%TYPE
) RETURN jobs.job_title%TYPE IS
    v_title jobs.job_title%TYPE;
BEGIN
    SELECT job_title INTO v_title 
    FROM jobs 
    WHERE job_id = p_job_id;
    
    RETURN v_title;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20001, 'Praca o podanym ID (' || p_job_id || ') nie istnieje.');
END get_job_title;
/

CREATE OR REPLACE FUNCTION get_annual_salary (
    p_emp_id IN employees.employee_id%TYPE
) RETURN NUMBER IS
    v_annual_salary NUMBER;
BEGIN
    SELECT (salary * 12) + (salary * NVL(commission_pct, 0))
    INTO v_annual_salary
    FROM employees 
    WHERE employee_id = p_emp_id;
    
    RETURN v_annual_salary;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN NULL; 
END get_annual_salary;
/

CREATE OR REPLACE FUNCTION format_phone (
    p_phone IN VARCHAR2
) RETURN VARCHAR2 IS
    v_dot_pos NUMBER;
BEGIN
    v_dot_pos := INSTR(p_phone, '.');
    
    IF v_dot_pos > 0 THEN
      
        RETURN '(' || SUBSTR(p_phone, 1, v_dot_pos - 1) || ')' || SUBSTR(p_phone, v_dot_pos);
    ELSE
        RETURN '(' || SUBSTR(p_phone, 1, 2) || ')' || SUBSTR(p_phone, 3);
    END IF;
END format_phone;
/

CREATE OR REPLACE FUNCTION format_string_case (
    p_str IN VARCHAR2
) RETURN VARCHAR2 IS
    v_len NUMBER;
BEGIN
    IF p_str IS NULL THEN RETURN NULL; END IF;
    
    v_len := LENGTH(p_str);
    IF v_len = 1 THEN RETURN UPPER(p_str); END IF;
    
    RETURN UPPER(SUBSTR(p_str, 1, 1)) || 
           LOWER(SUBSTR(p_str, 2, v_len - 2)) || 
           UPPER(SUBSTR(p_str, v_len, 1));
END format_string_case;
/

CREATE OR REPLACE FUNCTION pesel_to_date (
    p_pesel IN VARCHAR2
) RETURN VARCHAR2 IS
    v_year NUMBER;
    v_month NUMBER;
    v_day NUMBER;
    v_full_year NUMBER;
BEGIN
    IF LENGTH(p_pesel) != 11 THEN 
        RAISE_APPLICATION_ERROR(-20002, 'Nieprawidłowa długość numeru PESEL.');
    END IF;

    v_year := TO_NUMBER(SUBSTR(p_pesel, 1, 2));
    v_month := TO_NUMBER(SUBSTR(p_pesel, 3, 2));
    v_day := TO_NUMBER(SUBSTR(p_pesel, 5, 2));
 
    IF v_month > 20 AND v_month < 33 THEN
        v_full_year := 2000 + v_year;
        v_month := v_month - 20;
    ELSE
        v_full_year := 1900 + v_year;
    END IF;
    
    RETURN TO_CHAR(v_full_year, 'FM0000') || '-' || TO_CHAR(v_month, 'FM00') || '-' || TO_CHAR(v_day, 'FM00');
END pesel_to_date;
/

CREATE OR REPLACE FUNCTION get_country_stats (
    p_country_name IN VARCHAR2
) RETURN VARCHAR2 IS
    v_country_id countries.country_id%TYPE;
    v_emp_count NUMBER := 0;
    v_dept_count NUMBER := 0;
BEGIN
 
    SELECT country_id INTO v_country_id
    FROM countries
    WHERE UPPER(country_name) = UPPER(p_country_name);

    SELECT COUNT(*) INTO v_dept_count
    FROM departments d
    JOIN locations l ON d.location_id = l.location_id
    WHERE l.country_id = v_country_id;
   
    SELECT COUNT(*) INTO v_emp_count
    FROM employees e
    JOIN departments d ON e.department_id = d.department_id
    JOIN locations l ON d.location_id = l.location_id
    WHERE l.country_id = v_country_id;
    
    RETURN 'Kraj: ' || UPPER(p_country_name) || ' | Pracownicy: ' || v_emp_count || ' | Departamenty: ' || v_dept_count;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20003, 'Kraj o nazwie ' || p_country_name || ' nie istnieje w bazie.');
END get_country_stats;
/

CREATE TABLE archiwum_departamentow (
    id NUMBER,
    nazwa VARCHAR2(100),
    data_zamkniecia DATE,
    ostatni_manager VARCHAR2(150)
);

CREATE OR REPLACE TRIGGER trg_dept_archive
AFTER DELETE ON departments
FOR EACH ROW
DECLARE
    v_manager_name VARCHAR2(150);
BEGIN

    IF :OLD.manager_id IS NOT NULL THEN
        BEGIN
            SELECT first_name || ' ' || last_name INTO v_manager_name
            FROM employees WHERE employee_id = :OLD.manager_id;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN 
                v_manager_name := 'Nie znaleziono danych';
        END;
    ELSE
        v_manager_name := 'Brak managera';
    END IF;


    INSERT INTO archiwum_departamentow (id, nazwa, data_zamkniecia, ostatni_manager)
    VALUES (:OLD.department_id, :OLD.department_name, SYSDATE, v_manager_name);
END;
/

    FUNCTION get_job_title(p_job_id IN VARCHAR2) RETURN VARCHAR2 IS
        v_title jobs.job_title%TYPE;
    BEGIN
        SELECT job_title INTO v_title FROM jobs WHERE job_id = p_job_id;
        RETURN v_title;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN RAISE_APPLICATION_ERROR(-20001, 'Praca nie istnieje.');
    END get_job_title;

    FUNCTION get_annual_salary(p_emp_id IN NUMBER) RETURN NUMBER IS
        v_annual NUMBER;
    BEGIN
        SELECT (salary * 12) + (salary * NVL(commission_pct, 0)) INTO v_annual
        FROM employees WHERE employee_id = p_emp_id;
        RETURN v_annual;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN RETURN NULL;
    END get_annual_salary;

    FUNCTION format_phone(p_phone IN VARCHAR2) RETURN VARCHAR2 IS
        v_dot_pos NUMBER := INSTR(p_phone, '.');
    BEGIN
        IF v_dot_pos > 0 THEN RETURN '(' || SUBSTR(p_phone, 1, v_dot_pos - 1) || ')' || SUBSTR(p_phone, v_dot_pos);
        ELSE RETURN '(' || SUBSTR(p_phone, 1, 2) || ')' || SUBSTR(p_phone, 3);
        END IF;
    END format_phone;

    FUNCTION format_string_case(p_str IN VARCHAR2) RETURN VARCHAR2 IS
        v_len NUMBER;
    BEGIN
        IF p_str IS NULL THEN RETURN NULL; END IF;
        v_len := LENGTH(p_str);
        IF v_len = 1 THEN RETURN UPPER(p_str); END IF;
        RETURN UPPER(SUBSTR(p_str, 1, 1)) || LOWER(SUBSTR(p_str, 2, v_len - 2)) || UPPER(SUBSTR(p_str, v_len, 1));
    END format_string_case;

    FUNCTION pesel_to_date(p_pesel IN VARCHAR2) RETURN VARCHAR2 IS
        v_y NUMBER; v_m NUMBER; v_d NUMBER; v_fy NUMBER;
    BEGIN
        IF LENGTH(p_pesel) != 11 THEN RAISE_APPLICATION_ERROR(-20002, 'Błąd PESEL'); END IF;
        v_y := TO_NUMBER(SUBSTR(p_pesel, 1, 2)); v_m := TO_NUMBER(SUBSTR(p_pesel, 3, 2)); v_d := TO_NUMBER(SUBSTR(p_pesel, 5, 2));
        IF v_m > 20 THEN v_fy := 2000 + v_y; v_m := v_m - 20;
        ELSE v_fy := 1900 + v_y; END IF;
        RETURN TO_CHAR(v_fy, 'FM0000') || '-' || TO_CHAR(v_m, 'FM00') || '-' || TO_CHAR(v_d, 'FM00');
    END pesel_to_date;

        v_cid VARCHAR2(2); v_e NUMBER; v_d NUMBER;
    BEGIN
        SELECT country_id INTO v_cid FROM countries WHERE UPPER(country_name) = UPPER(p_country_name);
        SELECT COUNT(*) INTO v_d FROM departments d JOIN locations l ON d.location_id = l.location_id WHERE l.country_id = v_cid;
        SELECT COUNT(*) INTO v_e FROM employees e JOIN departments d ON e.department_id = d.department_id JOIN locations l ON d.location_id = l.location_id WHERE l.country_id = v_cid;
        RETURN 'Pracownicy: ' || v_e || ' | Departamenty: ' || v_d;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN RAISE_APPLICATION_ERROR(-20003, 'Brak kraju.');
    END get_country_stats;

CREATE OR REPLACE PACKAGE pkg_regions_crud IS
    PROCEDURE create_region(p_id IN NUMBER, p_name IN VARCHAR2);
    FUNCTION read_region_name(p_id IN NUMBER) RETURN VARCHAR2;
    FUNCTION read_region_id(p_name IN VARCHAR2) RETURN NUMBER;
    PROCEDURE update_region(p_id IN NUMBER, p_new_name IN VARCHAR2);
    PROCEDURE delete_region(p_id IN NUMBER);
END pkg_regions_crud;
/

CREATE OR REPLACE PACKAGE BODY pkg_regions_crud IS

    PROCEDURE create_region(p_id IN NUMBER, p_name IN VARCHAR2) IS
    BEGIN
        INSERT INTO regions (region_id, region_name) VALUES (p_id, p_name);
        DBMS_OUTPUT.PUT_LINE('Region utworzony.');
    END create_region;

    FUNCTION read_region_name(p_id IN NUMBER) RETURN VARCHAR2 IS
        v_name regions.region_name%TYPE;
    BEGIN
        SELECT region_name INTO v_name FROM regions WHERE region_id = p_id;
        RETURN v_name;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN RETURN 'Brak regionu o podanym ID';
    END read_region_name;

    FUNCTION read_region_id(p_name IN VARCHAR2) RETURN NUMBER IS
        v_id regions.region_id%TYPE;
    BEGIN
        SELECT region_id INTO v_id FROM regions WHERE UPPER(region_name) = UPPER(p_name);
        RETURN v_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN RETURN NULL;
    END read_region_id;


    PROCEDURE update_region(p_id IN NUMBER, p_new_name IN VARCHAR2) IS
    BEGIN
        UPDATE regions SET region_name = p_new_name WHERE region_id = p_id;
        IF SQL%ROWCOUNT = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Nie zaktualizowano, region nie istnieje.');
        ELSE
            DBMS_OUTPUT.PUT_LINE('Zaktualizowano region.');
        END IF;
    END update_region;

    PROCEDURE delete_region(p_id IN NUMBER) IS
    BEGIN
        DELETE FROM regions WHERE region_id = p_id;
        IF SQL%ROWCOUNT = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Brak regionu do usunięcia.');
        ELSE
            DBMS_OUTPUT.PUT_LINE('Usunięto region.');
        END IF;
    END delete_region;

END pkg_regions_crud;
/
